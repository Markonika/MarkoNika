cmake_minimum_required(VERSION 3.16)
project(mooring LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)

add_library(mooring STATIC src/catenary.cpp src/lumped_mass_cable.cpp)
target_include_directories(mooring PUBLIC include)
target_include_directories(mooring SYSTEM PUBLIC third_party)

enable_testing()
add_executable(mooring_tests tests/test_main.cpp tests/test_catenary.cpp)
target_link_libraries(mooring_tests PRIVATE mooring)
target_include_directories(mooring_tests SYSTEM PRIVATE third_party/doctest)
add_test(NAME mooring_tests COMMAND mooring_tests)
