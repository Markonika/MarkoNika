cmake_minimum_required(VERSION 3.16)
project(mooring LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)

add_library(mooring STATIC src/catenary.cpp src/lumped_mass_cable.cpp src/case_runner.cpp src/rigid_body.cpp src/coupled_runner.cpp src/paredes.cpp)
target_include_directories(mooring PUBLIC include)
target_include_directories(mooring SYSTEM PUBLIC third_party)

add_executable(mooring_run apps/mooring_run.cpp)
target_link_libraries(mooring_run PRIVATE mooring)
add_executable(mooring_platform apps/mooring_platform.cpp)
target_link_libraries(mooring_platform PRIVATE mooring)
add_executable(paredes_report apps/paredes_report.cpp)
target_link_libraries(paredes_report PRIVATE mooring)

enable_testing()
add_executable(mooring_tests tests/test_main.cpp tests/test_catenary.cpp tests/test_dynamics.cpp tests/test_hydro.cpp tests/test_chalmers.cpp tests/test_3d.cpp tests/test_points.cpp tests/test_platform.cpp tests/test_paredes.cpp)
target_link_libraries(mooring_tests PRIVATE mooring)
target_include_directories(mooring_tests SYSTEM PRIVATE third_party/doctest)
target_compile_definitions(mooring_tests PRIVATE MOORING_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
find_package(Threads REQUIRED)
target_link_libraries(mooring_tests PRIVATE Threads::Threads)
add_test(NAME mooring_tests COMMAND mooring_tests)
add_test(NAME chalmers_grid COMMAND mooring_tests --no-skip -tc=*Chalmers*full*grid*)
add_test(NAME paredes_validation COMMAND mooring_tests --no-skip -tc=*moored*buoy*test*7*validation*)
